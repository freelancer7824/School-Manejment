# ছবি নাম মোবাইল ইমেইল ভূমিকা স্ট্যাটাস অ্যাকশন
ডাটা লোড হচ্ছে...
✅ যে কোনো শিক্ষক বা অ্যাকাউন্ট্যান্ট তৈরি হলে স্বয়ংক্রিয়ভাবে তালিকায় যুক্ত হবে।

// 👇 আপনার লগইন পেজের signInWithEmailAndPassword এর পরে এই কোড যোগ করুন

import { getFirestore, doc, getDoc } from "https://www.gstatic.com/firebasejs/10.10.0/firebase-firestore.js";

const db = getFirestore();

async function handleLogin(email, password) {
    try {
        const userCredential = await signInWithEmailAndPassword(auth, email, password);
        const uid = userCredential.user.uid;

        const userDoc = await getDoc(doc(db, "users", uid));
        
        if (userDoc.exists()) {
            const userData = userDoc.data();
            
            if (userData.status === "inactive") {
                await auth.signOut();
                showToast("আপনার অ্যাকাউন্ট নিষ্ক্রিয় করা হয়েছে।", "error");
                return;
            }
            
            window.location.href = "dashboard.html";
        } else {
            await auth.signOut();
            showToast("ব্যবহারকারীর তথ্য পাওয়া যায়নি।", "error");
        }
    } catch (error) {
        showToast("লগইন ব্যর্থ: " + error.message, "error");
    }
}
